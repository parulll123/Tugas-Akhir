from flask import Flask, render_template, jsonify, request, redirect, url_for, flash
from flask_sqlalchemy import SQLAlchemy
from flask_login import LoginManager, UserMixin, login_user, login_required, logout_user, current_user
from werkzeug.security import generate_password_hash, check_password_hash
import datetime
import os

app = Flask(__name__)

# --- KONFIGURASI ---
app.config['SECRET_KEY'] = 'rahasia-skripsi-123'
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///users.db'
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False

db = SQLAlchemy(app)
login_manager = LoginManager()
login_manager.init_app(app)
login_manager.login_view = 'login'

# --- VARIABEL DATA & LOG (MEMORY) ---
current_motor_data = {
    "temperature": 0.0, "vibration": 0.0, "current_amp": 0.0,
    "status": "OFFLINE", "confidence": 0.0, "last_update": "-"
}

# List log riwayat anomali (Data Dummy Awal)
anomaly_logs = [
    {"time": "2025-01-10 10:15:00", "status": "DANGER", "vib": 7.2, "temp": 65.0},
    {"time": "2025-01-10 11:30:00", "status": "WARNING", "vib": 4.5, "temp": 50.0}
]

# Konfigurasi sistem
system_config = {
    "max_vib": 5.0,
    "max_temp": 80.0
}

# --- MODEL USER ---
class User(UserMixin, db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(150), unique=True, nullable=False)
    password = db.Column(db.String(150), nullable=False)

@login_manager.user_loader
def load_user(user_id):
    return User.query.get(int(user_id))

# --- ROUTES AUTH (LOGIN/LOGOUT) ---
@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')
        
        user = User.query.filter_by(username=username).first()
        
        if user and check_password_hash(user.password, password):
            login_user(user)
            return redirect(url_for('index'))
        else:
            flash('Login Gagal. Cek username dan password.', 'danger')
            
    return render_template('login.html')

@app.route('/logout')
@login_required
def logout():
    logout_user()
    return redirect(url_for('login'))

# --- ROUTES HALAMAN (VIEW) ---

@app.route('/')
@login_required
def index():
    # Halaman Dashboard Utama
    return render_template('index.html')

@app.route('/history')
@login_required
def history():
    # Halaman Riwayat, kirim data log yang di-reverse (terbaru di atas)
    return render_template('history.html', logs=reversed(anomaly_logs))

@app.route('/settings', methods=['GET', 'POST'])
@login_required
def config_page():
    # Halaman Konfigurasi
    if request.method == 'POST':
        try:
            system_config['max_vib'] = float(request.form.get('max_vib'))
            system_config['max_temp'] = float(request.form.get('max_temp'))
            flash('Konfigurasi berhasil diperbarui!', 'success')
        except ValueError:
            flash('Input harus berupa angka!', 'danger')
        
    return render_template('config.html', config_data=system_config)

# --- ROUTES API (DATA INTERCHANGE) ---

# 1. Menerima Data dari ESP32/Simulator
@app.route('/api/update', methods=['POST'])
def update_data():
    global current_motor_data
    try:
        data = request.json
        current_motor_data['temperature'] = data.get('temp', 0)
        current_motor_data['vibration'] = data.get('vib', 0)
        current_motor_data['current_amp'] = data.get('amp', 0)
        current_motor_data['status'] = data.get('status', 'UNKNOWN')
        current_motor_data['confidence'] = data.get('conf', 0)
        current_motor_data['last_update'] = datetime.datetime.now().strftime("%H:%M:%S")

        # LOGIKA SIMPAN LOG (Jika status DANGER)
        if data.get('status') == "DANGER":
            # Cek log terakhir agar tidak duplikat spam dalam detik yang sama
            now_str = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            
            # Tambahkan ke list log
            new_log = {
                "time": now_str,
                "status": "DANGER",
                "vib": data.get('vib'),
                "temp": data.get('temp')
            }
            anomaly_logs.append(new_log)
        
        return jsonify({"message": "Data received"}), 200
    except Exception as e:
        return jsonify({"error": str(e)}), 400

# 2. Mengirim Data ke Frontend (Polling JS)
@app.route('/api/live-data', methods=['GET'])
@login_required
def get_live_data():
    return jsonify(current_motor_data)
